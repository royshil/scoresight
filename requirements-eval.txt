# Evaluation-only extras for the OCR evaluation harness (ocr_eval/, see
# docs/development/ocr-eval.md): the pinned requirements of every engine
# adapter, grouped by engine. Each group is the adapter module's REQUIREMENTS
# (`python -m ocr_eval requirements <engine>` prints them, and
# tests/test_ocr_eval_requirements.py checks that this file matches).
# They are not app dependencies: install them into a SEPARATE virtual
# environment, together with the app's requirements (Tesseract and
# ScoreSight's own preprocessing come from there), e.g. on Windows:
#   uv venv --python 3.13 .venv-eval
#   uv pip install --python .venv-eval/Scripts/python.exe -r requirements-dev.txt -r requirements-win.txt -r requirements-eval.txt
# The Tesseract engine (tesseract_scoresight.py) needs nothing beyond
# requirements.txt.
-r requirements.txt

# --- PaddleOCR PP-OCRv6/v5 recognizers on onnxruntime (engines/ppocr.py) ---
# The models (Apache-2.0) are downloaded from the Hugging Face Hub at pinned
# revisions. onnxruntime 1.24 and later have no macOS x86_64 wheels (the last
# is 1.23.2, for macOS 13+); an Intel Mac gets that one (not measured).
onnxruntime==1.31.0 ; sys_platform != "darwin" or platform_machine != "x86_64"
onnxruntime==1.23.2 ; sys_platform == "darwin" and platform_machine == "x86_64"
# 1.x: OnnxTR 0.9.0 requires huggingface-hub < 2
huggingface_hub==2.2.0
PyYAML==6.0.3

# --- OnnxTR: docTR's PARSeq, ViTSTR, CRNN and VIPTR (engines/onnxtr.py) ---
